Para que esto funcione hoy mismo sin que tengas que montar un servidor propio, vamos a usar la infraestructura de Supabase (que es compatible al 100% con Netlify) para la base de datos, la seguridad y el inicio de sesión.

A. El Guardián de Seguridad (Auth.js)
Este script maneja el inicio de sesión y mantiene la sesión activa.

// Configuración de la conexión (Debes poner tus llaves de Supabase en Netlify)
import { createClient } from '@supabase/supabase-js'

const supabaseUrl = 'https://tu-proyecto.supabase.co'
const supabaseKey = process.env.SUPABASE_KEY
const supabase = createClient(supabaseUrl, supabaseKey)

// FUNCIÓN DE INICIO DE SESIÓN CON MEMORIA
async function loginUser(email, password) {
  const { data, error } = await supabase.auth.signInWithPassword({
    email: email,
    password: password,
  })
  
  if (error) {
    alert("Error al ingresar: " + error.message);
  } else {
    // Esto guarda la sesión en el navegador automáticamente
    window.location.href = "/dashboard"; 
  }
}

// PROTECCIÓN DE RUTA ADMINISTRADOR
function checkAdminAccess() {
  const session = supabase.auth.session();
  if (!session || session.user.user_metadata.role !== 'admin') {
    window.location.href = "/login?error=unauthorized";
  }
}

// Función para conectar personas con necesidades
async function buscarEspecialista(categoria) {
  let { data: especialistas, error } = await supabase
    .from('perfiles')
    .select('*')
    .eq('especialidad', categoria)
    .eq('esta_verificado', true);

  if (error) console.error("Error en búsqueda", error);
  
  // Renderizado dinámico (esto reemplaza tus iconos estáticos)
  const contenedor = document.getElementById('grid-especialistas');
  contenedor.innerHTML = especialistas.map(esp => `

      <h3>${esp.nombre}</h3>
      <p>${esp.bio}</p>
      <button onclick="conectar(${esp.id})">Contactar Ahora</button>
    </div>
  `).join('');
}

async function procesarMembresia(priceId) {
  const { error } = await stripe.redirectToCheckout({
    lineItems: [{ price: priceId, quantity: 1 }],
    mode: 'subscription',
    successUrl: 'https://www.centromultidisciplinarioags.com/success',

  if (error) console.error("Error en pago:", error);
